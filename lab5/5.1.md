#include <iostream>
#include <cstring>
using namespace std;

#define INF 9999999
#define V 8

int main() {
    int G[V][V] = {
        { INF, INF,   2,   4, INF,   8, INF, INF }, // 1
        { INF, INF, INF,   6,   7,   6, INF, INF }, // 2
        {   2, INF, INF, INF, INF,   3, INF,   4 }, // 3
        {   4,   6, INF, INF,   1, INF, INF, INF }, // 4
        { INF,   7, INF,   1, INF, INF,   4, INF }, // 5
        {   8,   6,   3, INF, INF, INF,   5,   5 }, // 6
        { INF, INF, INF, INF,   4,   5, INF, INF }, // 7
        { INF, INF,   4, INF, INF,   5, INF, INF }  // 8
    };

    int selected[V];
    memset(selected, false, sizeof(selected));
    
    selected[0] = true;
    int no_edge = 0;

    cout << "Edge : Weight" << endl;
    
    while (no_edge < V - 1) {
        int min = INF;
        int x = 0;
        int y = 0;

        for (int i = 0; i < V; i++) {
            if (selected[i]) {
                for (int j = 0; j < V; j++) {
                    if (!selected[j] && G[i][j]) { 
                        if (min > G[i][j]) {
                            min = G[i][j];
                            x = i;
                            y = j;
                        }
                    }
                }
            }
        }

        cout << x + 1 << " - " << y + 1 << " : " << G[x][y] << endl;
        
        selected[y] = true;
        no_edge++;
    }

    return 0;
}
