#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

#define edge pair<int,int>

class Graph {
private:
    vector<pair<int, edge>> G; // граф (вага, (вершина1, вершина2))
    vector<pair<int, edge>> T; // кістякове дерево (МКД)
    int *parent;
    int V; // кількість вершин
public:
    Graph(int V);
    void AddWeightedEdge(int u, int v, int w);
    int find_set(int i);
    void union_set(int u, int v);
    void kruskal();
    void print();
};

Graph::Graph(int V) {
    this->V = V;
    parent = new int[V];
    for (int i = 0; i < V; i++)
        parent[i] = i;
    G.clear();
    T.clear();
}

void Graph::AddWeightedEdge(int u, int v, int w) {
    G.push_back(make_pair(w, edge(u, v)));
}

int Graph::find_set(int i) {
    if (i == parent[i])
        return i;
    return parent[i] = find_set(parent[i]);
}

void Graph::union_set(int u, int v) {
    parent[u] = parent[v];
}

void Graph::kruskal() {
    int i, uRep, vRep;
    sort(G.begin(), G.end()); // сортування за зростанням ваги

    for (i = 0; i < G.size(); i++) {
        uRep = find_set(G[i].second.first);
        vRep = find_set(G[i].second.second);
        if (uRep != vRep) {
            T.push_back(G[i]); // додаємо до дерева
            union_set(uRep, vRep);
        }
    }
}

void Graph::print() {
    cout << "Edge :" << " Weight" << endl;
    for (int i = 0; i < T.size(); i++) {
        cout << T[i].second.first << " - " << T[i].second.second << " : " << T[i].first;
        cout << endl;
    }
}

int main() {
    // Створюємо граф на 9 елементів, щоб індекси вершин (1-8) збігалися з малюнком
    Graph g(9);

    // Додаємо ребра суворо за твоїм малюнком:
    g.AddWeightedEdge(1, 3, 2);
    g.AddWeightedEdge(1, 4, 4);
    g.AddWeightedEdge(1, 6, 8);
    g.AddWeightedEdge(2, 4, 6);
    g.AddWeightedEdge(2, 5, 7);
    g.AddWeightedEdge(2, 6, 6);
    g.AddWeightedEdge(3, 6, 3);
    g.AddWeightedEdge(3, 8, 4);
    g.AddWeightedEdge(4, 5, 1);
    g.AddWeightedEdge(5, 7, 4);
    g.AddWeightedEdge(6, 7, 5);
    g.AddWeightedEdge(6, 8, 5);

    g.kruskal();
    g.print();

    return 0;
}
