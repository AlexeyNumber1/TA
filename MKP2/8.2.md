StackDFS(G, s):
    Push(S, (s, ENTER))
    while not IsStackEmpty(S) do
        (u, state) <- Pop(S)
        if state = ENTER then
            if not visited[u] then
                visited[u] <- true
                timer <- timer + 1
                tin[u] <- timer
                if parent[u] != -1 then
                    E_T <- E_T U {(parent[u], u)}
                Push(S, (u, EXIT))
                for each v in Reverse(Adj[u]) do
                    if not visited[v] then
                        parent[v] <- u
                        Push(S, (v, ENTER))
        else
            timer <- timer + 1
            tout[u] <- timer
